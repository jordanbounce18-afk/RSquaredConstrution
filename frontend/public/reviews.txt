<script>
  let rating = 0;
  const starEls = document.querySelectorAll('#stars span');
  starEls.forEach(function (el) {
    el.addEventListener('click', function () {
      rating = parseInt(el.getAttribute('data-v'), 10);
      paintStars();
    });
  });
  function paintStars() {
    starEls.forEach(function (el) {
      var v = parseInt(el.getAttribute('data-v'), 10);
      el.classList.toggle('filled', v <= rating);
    });
  }

  var form = document.getElementById('reviewForm');
  var msg = document.getElementById('formMsg');
  var btn = document.getElementById('submitBtn');

  form.addEventListener('submit', function (e) {
    e.preventDefault();
    msg.textContent = '';
    msg.className = '';

    var name = document.getElementById('name').value.trim();
    var text = document.getElementById('text').value.trim();

    if (!name || !text || rating < 1) {
      msg.textContent = 'Please fill in your name, a review, and select a star rating.';
      msg.className = 'err';
      return;
    }

    btn.disabled = true;
    btn.textContent = 'Sending...';

    fetch('/api/reviews', {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({ name: name, rating: rating, text: text })
    })
      .then(function (r) { return r.ok ? r.json() : Promise.reject(); })
      .then(function () {
        msg.textContent = 'Thank you! Your review has been submitted and will appear here once approved.';
        msg.className = 'ok';
        form.reset();
        rating = 0;
        paintStars();
      })
      .catch(function () {
        msg.textContent = 'Something went wrong. Please try again.';
        msg.className = 'err';
      })
      .finally(function () {
        btn.disabled = false;
        btn.textContent = 'Submit Review';
      });
  });

  function loadReviews() {
    fetch('/api/reviews')
      .then(function (r) { return r.json(); })
      .then(function (data) {
        var list = document.getElementById('reviewsList');
        var empty = document.getElementById('reviewsEmpty');
        var reviews = (data && data.reviews) || [];
        if (reviews.length === 0) {
          empty.style.display = 'block';
          return;
        }
        list.innerHTML = reviews.map(function (r) {
          var stars = '★'.repeat(r.rating) + '☆'.repeat(5 - r.rating);
          var name = String(r.name || '').replace(/</g, '&lt;');
          var text = String(r.text || '').replace(/</g, '&lt;');
          return '<div class="review">' +
            '<div class="stars-static">' + stars + '</div>' +
            '<div class="name">' + name + '</div>' +
            '<div class="text">' + text + '</div>' +
          '</div>';
        }).join('');
      })
      .catch(function () {});
  }
  loadReviews();
</script>
